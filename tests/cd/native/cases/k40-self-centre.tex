% PGF's `\pgfpointshapeborder` writes `Returning node center instead of a point on node border` when it looks for the border towards the node's own centre.  An anchor on
% one end of an arrow from a node to itself does not stop that: the other end, which has none, still looks towards the first, and with `center` that is the centre.
% Only real PGF writes that warning, so those diagrams go to it.  With anchors on both ends, a loop's angles or another anchor, nothing looks towards the centre.

\usepackage{tikz-cd}
\begin{document}

\begin{tikzcd} B \arrow[loop, start anchor=center, bend left] \end{tikzcd}

\begin{tikzcd} C \arrow[to=1-1, end anchor=center, bend right=20] \end{tikzcd}

\begin{tikzcd} |[alias=Z]| D \arrow[from=Z, to=Z, start anchor=center, shift left] \end{tikzcd}

\begin{tikzcd} E \arrow[to=1-1, start anchor={[xshift=0pt]center}] \end{tikzcd}
