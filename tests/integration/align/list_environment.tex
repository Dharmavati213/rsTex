\setbox1\vbox{\prevdepth=3pt \hrule\halign{#\cr g\cr}\showthe\prevdepth \kern 1pt} \showbox1

\pdfignoreddimen=-500pt
\setbox1\vbox{\prevdepth=-500pt \halign{#\cr g\cr}\showthe\prevdepth} \showbox1
\setbox1\vbox{\prevdepth=-1000pt \halign{#\cr g\cr}\showthe\prevdepth} \showbox1
