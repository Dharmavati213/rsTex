  A \the\fontcharwd#1`A \the\fontcharht#1`A \the\fontchardp#1`A \the\fontcharic#1`A \the\tagcode#1`A;
  f \the\fontcharwd#1`f \the\fontcharht#1`f \the\fontchardp#1`f \the\fontcharic#1`f \the\tagcode#1`f;
  nul \the\fontcharwd#1 0 \the\tagcode#1 0;
  del \the\fontcharwd#1 127 \the\tagcode#1 127;
  top \the\fontcharwd#1 255 \the\tagcode#1 255]}}
