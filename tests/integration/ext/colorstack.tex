\edef\t{\pdfcolorstackinit direct{1 0 0 rg}}
\edef\u{\pdfcolorstackinit{0 1 0 rg}}
\message{[\s,\t,\u]}
\setbox1\hbox{\pdfcolorstack\s set{1 g}a\pdfcolorstack\s push{0.5 g}b\pdfcolorstack\s pop c\pdfcolorstack\s current d%
  \pdfcolorstack\t push{0 0 1 rg}\pdfcolorstack\t pop \pdfcolorstack\u set{0 0 1 rg}}
\showbox1
\setbox2\copy1 \setbox2\box9
\shipout\copy1

\shipout\hbox{\pdfcolorstack\s current}

\showbox2
\setbox2\hbox{\pdfcolorstack -1 push{x}}
\showbox2

\showbox2

\showbox2

\showbox2
