\pdfoutline goto file{x.pdf} struct{/s} page 3 {/Fit} newwindow {a}
\pdfoutline goto file{(x.pdf)} struct{(s)} name{n} nonewwindow {b}
\pdfoutline goto struct name{s} num 4 {c}
\pdfoutline goto struct num 7 name{n} {d}
\pdfoutline thread file{(y.pdf)} num 2 {e}
\pdfoutline thread file{y.pdf} name{t} {f}
\shipout\hbox{\pdfdest num 4 fit \pdfdest struct 7 num 4 fit}
