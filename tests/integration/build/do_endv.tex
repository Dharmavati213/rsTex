% Aimed at do_endv's walk with rec_at_list_end: the end of every v-template of every cell, reached with levels above it that
% are at their end and pass (macros, \the, inserted, backed-up), both on the walk's first test and at the bottom, in \halign,

% has been read; and cells that end in macros.

\halign{#\m\cr \m\cr}\showlists\reset
\halign{#\mmm\cr \mmm\cr}\showlists\reset

\halign{#\hskip1pt\cr a\cr}\showlists\reset
\halign{\hbox{#}\cr a\cr}\showlists\reset
\halign{$#$\cr a\cr}\showlists\reset
% \omit and \span.
\halign{#&#\cr \omit a&\omit b\cr}\showlists\reset
\halign{#&#&#\cr a\span b\span c\cr \omit\span\omit\span\omit x\cr}\showlists\reset

\setbox0\hbox{\valign{#\m\cr\def\m{\hrule}\cr}}\showbox0
$$\halign{#\cr a\cr}\showlists$$\reset
