% mode: latex
% env: RSTEX_CASE_MEMO_EAGER=1
%% takes: 60
% The case change of texts with every printable ASCII character, groups, \protect, math, accents and UTF-8, in each of its forms: each three times in a
% message and twice as \MakeUppercase in the text, which crate::case_memo records once and replays after.  \tracingstats shows the stacks' marks.
\documentclass{article}
\tracingstats=2
\ExplSyntaxOn
\cs_new_protected:Npn \casetest #1
  {
    \prg_replicate:nn { 3 }
      {
        \iow_term:e { U[ \text_uppercase:n {#1} ] }
        \iow_term:e { L[ \text_lowercase:n {#1} ] }
        \iow_term:e { T[ \text_titlecase_first:n {#1} ] }
        \iow_term:e { A[ \text_titlecase_all:n {#1} ] }
      }
  }
\cs_new_protected:Npn \casestr
  {
    \casetest
      {
        \c_hash_str \c_dollar_str \c_percent_str \c_ampersand_str \c_circumflex_str \c_underscore_str \c_tilde_str
        \c_left_brace_str \c_right_brace_str \c_backslash_str
      }
  }
\ExplSyntaxOff
