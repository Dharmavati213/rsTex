% fnt2 and fnt_def2.  So 300 fonts, each used.  Each font's definition comes once, in the page of its first use, and in the
% postamble; dvi_f is reset by each page; a font with an area name; a font loaded but never used.

\pdfoutput=0 \year=2026 \month=11 \day=2 \time=754
\tracingonline=0 \hbadness=10000 \vbadness=10000 \hfuzz=16383pt \vfuzz=16383pt
\countdef\n=10 \n=0
\def\loop{\advance\n by 1
  \expandafter\font\csname f\number\n\endcsname=cmr10 at \number\n sp
  \ifnum\n<300 \expandafter\loop\fi}
\def\use#1{\csname f#1\endcsname}
\loop
% a page using some, out of order; another with the same
\shipout\hbox{\use{1}a\use{2}a\use{1}a\use{64}a\use{65}a\use{64}a\use{255}a\use{256}a\use{257}a\use{258}a\use{300}a}
\shipout\hbox{\use{1}a\use{2}a\use{1}a\use{64}a\use{65}a\use{64}a\use{255}a\use{256}a\use{257}a\use{258}a\use{300}a}
% all of them, in order, in a vbox of hboxes, with one per line: definitions for those not yet used
\n=0
\def\line{\advance\n by 1 \hbox{\use{\number\n}a}\ifnum\n<300 \expandafter\line\fi}
\shipout\vbox{\line}

\font\area=./cmr10 \font\ext=cmr10.tfm \font\scaled=cmr10 scaled 1200 \font\sized=cmr10 at 7pt
