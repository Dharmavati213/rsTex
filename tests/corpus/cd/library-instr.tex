\def\cdlib@install{%

  \def\pgfmultipartnode##1##2##3##4{%

    \cdlib@mpn{##1}{##2}{##3}{##4}%

    \immediate\write-1{CDPIC|\the\pgf@picminx|\the\pgf@picminy|\the\pgf@picmaxx|\the\pgf@picmaxy|\pgf@shift@baseline}%
    \cdlib@tpb##1%
  }%

      $\relax$%
      \pgfmathsetlength\pgf@x{1ex}\edef\cd@ex{\the\pgf@x}%

    \endgroup
  }}%
}
